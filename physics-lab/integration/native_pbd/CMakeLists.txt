cmake_minimum_required(VERSION 3.20)
project(LiminalNativePbdShot LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(ovphysx REQUIRED)
add_executable(liminal_native_pbd_shot main.c)
target_include_directories(liminal_native_pbd_shot PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/ovphysx-sdk/ovphysx/samples/c_samples/common")
target_link_libraries(liminal_native_pbd_shot PRIVATE ovphysx::ovphysx ovphysx::ovstage)
if(WIN32)
  ovphysx_copy_runtime_dlls(liminal_native_pbd_shot)
  ovphysx_set_ovstage_runtime_env(liminal_native_pbd_shot)
endif()
